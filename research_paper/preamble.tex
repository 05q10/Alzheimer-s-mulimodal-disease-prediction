% Image placeholder macro
% Usage: \figph{filename_without_extension}{caption_text}{label_text}
\newcommand{\figph}[3]{%
  \begin{figure}[H]
    \centering
    \IfFileExists{figures/#1.pdf}{%
      \includegraphics[width=0.88\linewidth]{figures/#1.pdf}%
    }{%
      \IfFileExists{figures/#1.png}{%
        \includegraphics[width=0.88\linewidth]{figures/#1.png}%
      }{%
        \framebox[\linewidth]{\rule{0pt}{140pt}\Large\textbf{PLACEHOLDER: figures/#1}}%
      }%
    }%
    \caption{#2}
    \label{#3}
  \end{figure}%
}
